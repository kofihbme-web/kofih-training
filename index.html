📝 Apps Script Code — Copy & Paste into Extensions → Apps Script

function doGet(e) {
  var sheet   = SpreadsheetApp.getActiveSpreadsheet()
                .getSheetByName("Participants");
  var data    = sheet.getDataRange().getValues();
  var headers = data[0];
  var tel     = e.parameter.tel;
  var all     = e.parameter.all;

  if (tel) {
    // Return single participant by telephone (Participant tab lookup)
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(tel).trim()) {
        var row = {};
        headers.forEach(function(h, idx) { row[h] = data[i][idx]; });
        return out({ status: "ok", participant: row });
      }
    }
    return out({ status: "not_found" });
  }

  // Build stats + optional full list
  // A row counts as "completed" if:
  //   - the status column says "completed", OR
  //   - a score value exists (not blank) in the score column
  var stats = { total: data.length - 1,
                selfPre: 0, facPre: 0,
                selfPost: 0, facPost: 0, exam: 0 };
  var participants = [];

  for (var j = 1; j < data.length; j++) {
    var r = {};
    headers.forEach(function(h, idx) { r[h] = data[j][idx]; });

    // Helper: count as completed if status = "completed" OR score cell has a value
    if (isDone(r.selfPre,  r.selfPreScore))  stats.selfPre++;
    if (isDone(r.facPre,   r.facPreScore))   stats.facPre++;
    if (isDone(r.selfPost, r.selfPostScore)) stats.selfPost++;
    if (isDone(r.facPost,  r.facPostScore))  stats.facPost++;
    if (isDone(r.exam,     r.examScore))     stats.exam++;

    if (all === "1") { participants.push(r); }
  }

  if (all === "1") {
    return out({ status: "ok", stats: stats, participants: participants });
  }
  return out({ status: "ok", stats: stats });
}

// Count as done if status = "completed" OR a numeric score exists
function isDone(status, score) {
  if (String(status).trim().toLowerCase() === "completed") return true;
  if (score !== "" && score !== null && score !== undefined && !isNaN(parseFloat(score))) return true;
  return false;
}

function out(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// doPost: receives daily report updates from the portal Facilitator tab
function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    if (payload.action !== 'updateReports') {
      return out({ status: 'error', message: 'Unknown action' });
    }
    var sheet   = SpreadsheetApp.getActiveSpreadsheet()
                  .getSheetByName("Participants");
    var data    = sheet.getDataRange().getValues();
    var headers = data[0];
    var colIdx  = {};
    headers.forEach(function(h, i) { colIdx[h] = i; });

    var telCol = colIdx['telephone'];
    var dayKeys = ['day2Report','day3Report','day4Report','day5Report',
                   'day6Report','day7Report','day8Report','day9Report'];

    payload.reports.forEach(function(rep) {
      var tel = String(rep.telephone).trim();
      for (var r = 1; r < data.length; r++) {
        if (String(data[r][telCol]).trim() === tel) {
          dayKeys.forEach(function(key) {
            if (colIdx[key] !== undefined && rep[key] !== undefined) {
              sheet.getRange(r + 1, colIdx[key] + 1).setValue(rep[key]);
            }
          });
          break;
        }
      }
    });
    return out({ status: 'ok' });
  } catch(err) {
    return out({ status: 'error', message: err.toString() });
  }
}
